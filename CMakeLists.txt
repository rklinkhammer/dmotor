cmake_minimum_required(VERSION 3.16)
project(dmotor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DMOTOR_SANITIZE "Enable undefined behavior sanitizer" ON)
add_library(dmotor src/protocol.cpp src/arithmetic.cpp)
target_include_directories(dmotor PUBLIC include)
target_compile_options(dmotor PRIVATE -Wall -Wextra -Werror -Wconversion)
if(DMOTOR_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dmotor PUBLIC -fsanitize=undefined -fno-sanitize-recover=all)
  target_link_options(dmotor PUBLIC -fsanitize=undefined)
endif()
enable_testing()
add_executable(arithmetic_test tests/arithmetic_test.cpp)
target_link_libraries(arithmetic_test PRIVATE dmotor)
add_test(NAME arithmetic COMMAND arithmetic_test)
add_executable(protocol_test tests/protocol_test.cpp)
target_link_libraries(protocol_test PRIVATE dmotor)
add_test(NAME protocol COMMAND protocol_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME generated_abi COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/generate_abi.py --check)
find_program(IVERILOG iverilog)
find_program(VVP vvp)
set(IVERILOG_BASE "" CACHE PATH "Optional relocated Icarus backend directory")
if(IVERILOG AND VVP)
  foreach(test cleanup_sequencer monitor_heartbeat safety scheduler qcencoder pwm latency axi_bridge encoder_snapshot target_age shared_scheduler vivado_dev)
    add_test(NAME rtl_${test} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/rtl_test.py
      --iverilog ${IVERILOG} --vvp ${VVP} --base "${IVERILOG_BASE}"
      --source ${CMAKE_SOURCE_DIR} --build ${CMAKE_BINARY_DIR} --test ${test})
  endforeach()
else()
  message(WARNING "Icarus not found: RTL tests unavailable (not passed)")
endif()
