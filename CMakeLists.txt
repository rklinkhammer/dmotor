cmake_minimum_required(VERSION 3.16)
project(dmotor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DMOTOR_SANITIZE "Enable undefined behavior sanitizer" ON)
add_library(dmotor src/protocol.cpp src/arithmetic.cpp)
target_include_directories(dmotor PUBLIC include)
target_compile_options(dmotor PRIVATE -Wall -Wextra -Werror -Wconversion)
if(DMOTOR_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dmotor PUBLIC -fsanitize=undefined -fno-sanitize-recover=all)
  target_link_options(dmotor PUBLIC -fsanitize=undefined)
endif()
enable_testing()
add_executable(arithmetic_test tests/arithmetic_test.cpp)
target_link_libraries(arithmetic_test PRIVATE dmotor)
add_test(NAME arithmetic COMMAND arithmetic_test)
add_executable(protocol_test tests/protocol_test.cpp)
target_link_libraries(protocol_test PRIVATE dmotor)
add_test(NAME protocol COMMAND protocol_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME generated_abi COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/generate_abi.py --check)
find_program(IVERILOG iverilog)
find_program(VVP vvp)
set(IVERILOG_BASE "" CACHE PATH "Optional relocated Icarus backend directory")
if(IVERILOG AND VVP)
  foreach(test safety scheduler qcencoder pwm latency axi_bridge encoder_snapshot)
    add_test(NAME rtl_${test} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/rtl_test.py
      --iverilog ${IVERILOG} --vvp ${VVP} --base "${IVERILOG_BASE}"
      --source ${CMAKE_SOURCE_DIR} --build ${CMAKE_BINARY_DIR} --test ${test})
  endforeach()
else()
  message(WARNING "Icarus not found: RTL tests unavailable (not passed)")
endif()
target_sources(dmotor PRIVATE src/control.cpp)
add_executable(control_test tests/control_test.cpp)
target_link_libraries(control_test PRIVATE dmotor)
add_test(NAME control COMMAND control_test)
target_sources(dmotor PRIVATE src/command_engine.cpp)
add_executable(command_engine_test tests/command_engine_test.cpp)
target_link_libraries(command_engine_test PRIVATE dmotor)
add_test(NAME command_engine COMMAND command_engine_test)
target_sources(dmotor PRIVATE src/peripheral.cpp src/configuration.cpp)
add_executable(peripheral_test tests/peripheral_test.cpp)
target_link_libraries(peripheral_test PRIVATE dmotor)
add_test(NAME peripheral COMMAND peripheral_test)
# Requires the generated kernel: run tools/vitis_check.py --kernel peripheral.
if(IVERILOG AND VVP AND EXISTS "${CMAKE_SOURCE_DIR}/build/vitis_peripheral/hls/syn/verilog/dmotor_peripheral.v")
  add_test(NAME rtl_peripheral_integrated COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tools/integrated_rtl.py --iverilog ${IVERILOG} --vvp ${VVP}
    --base "${IVERILOG_BASE}")
endif()
add_executable(configuration_test tests/configuration_test.cpp)
target_link_libraries(configuration_test PRIVATE dmotor)
add_test(NAME configuration COMMAND configuration_test)
target_sources(dmotor PRIVATE src/motion.cpp)
add_executable(motion_test tests/motion_test.cpp)
target_link_libraries(motion_test PRIVATE dmotor)
add_test(NAME motion COMMAND motion_test)
add_executable(motion_queue_test tests/motion_queue_test.cpp)
target_link_libraries(motion_queue_test PRIVATE dmotor)
add_test(NAME motion_queue COMMAND motion_queue_test)
# Generated motion RTL plus autonomous scheduler and logical PWM adapter.
if(IVERILOG AND VVP AND EXISTS "${CMAKE_SOURCE_DIR}/build/vitis_motion/hls/syn/verilog/dmotor_motion.v")
  add_test(NAME rtl_motion_integrated COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tools/motion_rtl.py --iverilog ${IVERILOG} --vvp ${VVP}
    --base "${IVERILOG_BASE}")
endif()

target_sources(dmotor PRIVATE src/velocity.cpp)
add_executable(velocity_test tests/velocity_test.cpp)
target_link_libraries(velocity_test PRIVATE dmotor)
add_test(NAME velocity COMMAND velocity_test)
target_sources(dmotor PRIVATE src/controller.cpp)
add_executable(controller_test tests/controller_test.cpp)
target_link_libraries(controller_test PRIVATE dmotor)
add_test(NAME controller COMMAND controller_test)
target_sources(dmotor PRIVATE src/dispatcher.cpp)
add_executable(dispatcher_test tests/dispatcher_test.cpp)
target_link_libraries(dispatcher_test PRIVATE dmotor)
add_test(NAME dispatcher COMMAND dispatcher_test)
add_executable(position_error_test tests/position_error_test.cpp)
target_link_libraries(position_error_test PRIVATE dmotor)
add_test(NAME position_error COMMAND position_error_test)
